### 1. Verification Summary & Failsafe Execution Architecture

When an autonomous AI agent exhibits runaway or destructive execution (e.g. attempting recursive deletions, exfiltrating code, or looping in bash), the remote developer requires an **instant, deterministic panic switch** from mobile:

1. **Atomic Process Group Freezing via Negative PGID (`-PGID`)**:
   - Sending a signal to a single PID (`kill $PID`) leaves child processes (`node`, `git`, `bash`, `python`) running as unmonitored orphans.
   - When `bimaxd` launches the agent CLI or child processes, it sets `syscall.SysProcAttr{Setpgid: true}`, making the spawned process the leader of a fresh POSIX process group. The child’s PID becomes the process group ID (`pgid = cmd.Process.Pid`).
   - Calling `syscall.Kill(-pgid, syscall.SIGSTOP)` immediately halts every descendant process **instantaneously across all CPU cores** in $0\text{ ms}$.
2. **Two-Stage Termination (SIGTERM $\rightarrow$ SIGKILL)**:
   - For emergency shutdowns, `bimaxd` sends `syscall.Kill(-pgid, syscall.SIGTERM)` to allow critical subsystems (like `cloudflared` tunnels) to unregister cleanly and flush pending logs.
   - After a configurable short grace period (e.g., $500\text{ ms}$ to $2\text{ s}$), `bimaxd` fires `syscall.Kill(-pgid, syscall.SIGKILL)` as the unignorable hard stop.
3. **Modern macOS Lock Screen Without Sudo**:
   - Legacy scripts calling `/System/Library/CoreServices/Menu Extras/User.menu/Contents/Resources/CGSession -suspend` **fail on modern macOS releases (Sonoma / Sequoia)** because Apple removed this binary.
   - **Primary Deterministic Method**: A lightweight C helper compiled against `login.framework` invoking `SACLockScreenImmediate()`. It locks the screen immediately without logging out the user and without root privileges.
   - **Safe Public Fallback**: `/usr/bin/pmset displaysleepnow`, which immediately puts the display to sleep and engages the lock screen (when "Require password immediately after sleep" is active).
4. **Graceful Cloudflare Tunnel Shutdown & Diagnostics**:
   - Captures active tunnel health and diagnostics via `cloudflared tunnel diag --no-diag-network` before shutdown.
   - Runs `cloudflared tunnel cleanup <tunnel_id>` to purge stale connectors from the Cloudflare global edge.
5. **Structured JSON Incident Audit Log**:
   - Emits an append-only, forensic JSON record into `~/Library/Logs/bimax/incidents.jsonl` detailing timestamps, signals sent, tunnel status, and process snapshots.

### 2. Lock Screen Implementation: `SACLockScreenImmediate` & `pmset`

#### Primary Method: C Helper (`lockscreen.c`)
The private `login.framework` exports `SACLockScreenImmediate()`, which locks the active macOS user session immediately without root:

```c
// lockscreen.c
// Compile: clang -F /System/Library/PrivateFrameworks -framework login -o /usr/local/bin/lockscreen-helper lockscreen.c
extern int SACLockScreenImmediate(void);

int main(void) {
    return SACLockScreenImmediate();
}
```

#### Go Lock Screen Controller:
```go
package supervisor

import (
	"log"
	"os/exec"
)

// LockMacScreen executes the fastest available user-mode lock mechanism
func LockMacScreen() error {
	// 1. Primary: Dedicated lockscreen-helper binary (SACLockScreenImmediate)
	if _, err := exec.LookPath("/usr/local/bin/lockscreen-helper"); err == nil {
		cmd := exec.Command("/usr/local/bin/lockscreen-helper")
		if err := cmd.Run(); err == nil {
			log.Println("[PANIC] Display locked via SACLockScreenImmediate helper")
			return nil
		}
	}

	// 2. Fallback: Native unprivileged pmset displaysleepnow
	log.Println("[PANIC] Falling back to pmset displaysleepnow")
	cmd := exec.Command("/usr/bin/pmset", "displaysleepnow")
	return cmd.Run()
}
```

### 3. Complete Production Panic Controller in Go (`panic_controller.go`)

```go
package supervisor

import (
	"encoding/json"
	"fmt"
	"log"
	"net"
	"os"
	"os/exec"
	"path/filepath"
	"sync"
	"syscall"
	"time"
)

type PanicState int

const (
	StateNormal PanicState = iota
	StateFrozen
	StateTerminating
	StateTerminated
)

type IncidentRecord struct {
	Timestamp        string   `json:"timestamp"`
	AgentID          string   `json:"agent_id"`
	PGID             int      `json:"pgid"`
	Action           string   `json:"action"` // "freeze" | "terminate"
	SignalsSent      []string `json:"signals_sent"`
	LockScreenMethod string   `json:"lock_screen_method"`
	CloudflareTunnel struct {
		Name       string `json:"name"`
		CleanupRun bool   `json:"cleanup_run"`
		DiagZip    string `json:"diag_zip,omitempty"`
	} `json:"cloudflare"`
	Reason    string `json:"reason"`
	Initiator string `json:"initiator"` // "user" | "policy_breach"
}

type PanicController struct {
	mu           sync.Mutex
	state        PanicState
	agentID      string
	agentPGID    int
	tunnelName   string
	netConn      net.Conn
	incidentDir  string
}

func NewPanicController(agentID string, pgid int, tunnelName string, conn net.Conn) *PanicController {
	homeDir, _ := os.UserHomeDir()
	logDir := filepath.Join(homeDir, "Library", "Logs", "bimax")
	_ = os.MkdirAll(logDir, 0700)

	return &PanicController{
		state:       StateNormal,
		agentID:     agentID,
		agentPGID:   pgid,
		tunnelName:  tunnelName,
		netConn:     conn,
		incidentDir: logDir,
	}
}

// 1. FreezeExecution halts the entire process group immediately via SIGSTOP
func (pc *PanicController) FreezeExecution(reason, initiator string) error {
	pc.mu.Lock()
	defer pc.mu.Unlock()

	if pc.agentPGID <= 0 {
		return fmt.Errorf("invalid PGID: %d", pc.agentPGID)
	}

	log.Printf("[PANIC] Freezing agent process group -%d via SIGSTOP (Reason: %s)", pc.agentPGID, reason)
	err := syscall.Kill(-pc.agentPGID, syscall.SIGSTOP)
	if err == nil {
		pc.state = StateFrozen
		_ = LockMacScreen()
		pc.recordIncident("freeze", []string{"SIGSTOP"}, "SACLockScreenImmediate", reason, initiator, "")
	}
	return err
}

// 2. ResumeExecution unfreezes the paused processes via SIGCONT
func (pc *PanicController) ResumeExecution() error {
	pc.mu.Lock()
	defer pc.mu.Unlock()

	if pc.agentPGID <= 0 {
		return fmt.Errorf("invalid PGID: %d", pc.agentPGID)
	}

	log.Printf("[RESUME] Resuming agent process group -%d via SIGCONT", pc.agentPGID)
	err := syscall.Kill(-pc.agentPGID, syscall.SIGCONT)
	if err == nil {
		pc.state = StateNormal
	}
	return err
}

// 3. TerminateAndLockdown executes two-stage teardown, captures diag, severs network, and locks screen
func (pc *PanicController) TerminateAndLockdown(reason, initiator string, gracePeriod time.Duration) error {
	pc.mu.Lock()
	defer pc.mu.Unlock()

	log.Printf("[NUCLEAR PANIC] Initiating full system lockdown! Initiator: %s", initiator)
	pc.state = StateTerminating
	signals := []string{}

	// Step A: Lock physical macOS screen immediately
	_ = LockMacScreen()

	// Step B: Collect cloudflared diagnostic bundle before killing processes
	diagZip := pc.collectTunnelDiagnostics()

	// Step C: Send SIGTERM to process group for graceful unregister
	if pc.agentPGID > 0 {
		log.Printf("[PANIC] Sending SIGTERM to process group -%d", pc.agentPGID)
		_ = syscall.Kill(-pc.agentPGID, syscall.SIGTERM)
		signals = append(signals, "SIGTERM")

		// Short grace period before hard force-kill
		time.Sleep(gracePeriod)

		// Step D: Send SIGKILL to guarantee zero surviving processes
		log.Printf("[PANIC] Sending SIGKILL to process group -%d", pc.agentPGID)
		_ = syscall.Kill(-pc.agentPGID, syscall.SIGKILL)
		signals = append(signals, "SIGKILL")
		pc.state = StateTerminated

		// Reap zombie child processes
		go pc.reapZombies(pc.agentPGID)
	}

	// Step E: Sever active WebSocket tunnel network socket
	if pc.netConn != nil {
		_ = pc.netConn.Close()
	}

	// Step F: Explicit Cloudflare tunnel edge cleanup
	if pc.tunnelName != "" {
		_ = exec.Command("cloudflared", "tunnel", "cleanup", pc.tunnelName).Run()
	}

	// Step G: Write structured JSON incident record to disk
	pc.recordIncident("terminate", signals, "SACLockScreenImmediate", reason, initiator, diagZip)

	return nil
}

func (pc *PanicController) collectTunnelDiagnostics() string {
	timestamp := time.Now().Format("20060102-150405")
	diagFile := filepath.Join(pc.incidentDir, fmt.Sprintf("cloudflared-diag-%s.zip", timestamp))

	cmd := exec.Command("cloudflared", "tunnel", "diag", "--no-diag-network", "--output", diagFile)
	cmd.Dir = pc.incidentDir
	_ = cmd.Run()
	return diagFile
}

func (pc *PanicController) recordIncident(action string, signals []string, lockMethod, reason, initiator, diagZip string) {
	rec := IncidentRecord{
		Timestamp:        time.Now().UTC().Format(time.RFC3339Nano),
		AgentID:          pc.agentID,
		PGID:             pc.agentPGID,
		Action:           action,
		SignalsSent:      signals,
		LockScreenMethod: lockMethod,
		Reason:           reason,
		Initiator:        initiator,
	}
	rec.CloudflareTunnel.Name = pc.tunnelName
	rec.CloudflareTunnel.CleanupRun = (action == "terminate")
	rec.CloudflareTunnel.DiagZip = diagZip

	jsonBytes, err := json.Marshal(rec)
	if err == nil {
		incidentPath := filepath.Join(pc.incidentDir, "incidents.jsonl")
		f, err := os.OpenFile(incidentPath, os.O_APPEND|os.O_CREATE|os.O_WRONLY, 0600)
		if err == nil {
			defer f.Close()
			f.WriteString(string(jsonBytes) + "\n")
		}
	}
}

func (pc *PanicController) reapZombies(pgid int) {
	for {
		var wstatus syscall.WaitStatus
		pid, err := syscall.Wait4(-pgid, &wstatus, syscall.WNOHANG, nil)
		if pid <= 0 || err != nil {
			break
		}
		log.Printf("[REAPER] Reaped child PID %d", pid)
	}
}
```
