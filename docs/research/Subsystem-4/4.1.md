### 1. Verification Summary & PTY Architectural Principles

Autonomous developer tools like **Claude Code**, interactive shells (`zsh`, `bash`), and CLI utilities require a true POSIX **Pseudo-Terminal (PTY)** rather than standard redirected OS pipes (`os.Pipe`). Spawning interactive CLIs with standard pipes breaks functionality in three critical ways:
1. ANSI cursor positioning, interactive progress spinners, syntax colors, and raw line editing fail completely.
2. The CLI detects non-interactive TTY mode (`isatty(3) == false`) and switches into batch mode or disables interactive prompts.
3. Signal handling (`SIGINT`, `SIGWINCH`, `SIGTSTP`) fails to propagate across the sub-process hierarchy.

By utilizing **`github.com/creack/pty`** on macOS:
* The Go supervisor daemon creates a virtual master/slave PTY pair (`/dev/ptmx`).
* Claude Code runs in raw TTY mode, preserving full 256-color and 24-bit TrueColor ANSI styling.
* Terminal resize events triggered from the Android mobile client are translated into POSIX `ioctl(TIOCSWINSZ)` system calls, dynamically firing `SIGWINCH` to Claude Code so it redraws its interface to match the phone's screen width.
* The daemon runs a non-destructive, sliding-window pattern matcher on a **copy** of the output stream to detect confirmation prompts (e.g., `[y/n]`, `Do you want to proceed?`) without corrupting or delaying the raw ANSI stream streamed to mobile.

```
+-----------------------------------------------------------------------------------------+
|                                  BIMAXD PTY ARCHITECTURE                                |
|                                                                                         |

|         +--------------------------------------------+-----------------------------+    |
|         | Master PTY (/dev/ptmx)                                                   |    |
|         |                                                                          |    |
|         |  1. Inbound Keystrokes / Approvals: Write([]byte)                        |    |
```

### 2. Complete PTY Daemon Implementation in Go (`pty_daemon.go`)

```go
// +build darwin,!ios

package agent

import (
	"bufio"
	"bytes"
	"context"
	"fmt"
	"log"
	"os"
	"os/exec"
	"os/signal"
	"regexp"
	"sync"
	"syscall"
	"time"

	"github.com/creack/pty"
)

type PTYDaemon struct {
	cmd        *exec.Cmd
	ptyMaster  *os.File
	ctx        context.Context
	cancel     context.CancelFunc
	outCh      chan []byte      // Raw ANSI chunks forwarded to mobile client
	promptCh   chan string      // Detected confirmation prompt notifications
	mu         sync.Mutex
	buf        bytes.Buffer     // Sliding window buffer for prompt detection
	maxBufSize int
	pgid       int
}

// NewPTYDaemon spawns a program (e.g., Claude Code or zsh) inside a POSIX PTY
func NewPTYDaemon(program string, args []string, workingDir string, rows, cols uint16) (*PTYDaemon, error) {
	cmd := exec.Command(program, args...)
	if workingDir != "" {
		cmd.Dir = workingDir
	}

	// Crucial: Create a distinct Process Group ID (PGID) for atomic freeze/kill signals
	cmd.SysProcAttr = &syscall.SysProcAttr{
		Setpgid: true,
	}

	// Ensure terminal environment variables support rich styling
	cmd.Env = append(os.Environ(),
		"TERM=xterm-256color",
		"COLORTERM=truecolor",
		"FORCE_COLOR=1",
	)

	ws := &pty.Winsize{
		Rows: rows,
		Cols: cols,
		X:    0,
		Y:    0,
	}

	ptmx, err := pty.StartWithSize(cmd, ws)
	if err != nil {
		return nil, fmt.Errorf("failed to start pty: %w", err)
	}

	// Retrieve PGID for process group management
	pgid, err := syscall.Getpgid(cmd.Process.Pid)
	if err != nil {
		pgid = cmd.Process.Pid
	}

	ctx, cancel := context.WithCancel(context.Background())
	d := &PTYDaemon{
		cmd:        cmd,
		ptyMaster:  ptmx,
		ctx:        ctx,
		cancel:     cancel,
		outCh:      make(chan []byte, 128),
		promptCh:   make(chan string, 16),
		maxBufSize: 4096, // 4 KB sliding window
		pgid:       pgid,
	}

	go d.readLoop()
	go d.waitLoop()

	log.Printf("[bimaxd] PTY session spawned successfully (PID: %d, PGID: %d)", cmd.Process.Pid, pgid)
	return d, nil
}

// readLoop reads raw ANSI bytes from the PTY master and forwards them.
// It simultaneously feeds a side buffer for real-time prompt detection.
func (d *PTYDaemon) readLoop() {
	reader := bufio.NewReader(d.ptyMaster)

	for {
		select {
		case <-d.ctx.Done():
			return
		default:
			// Read small chunks (1024 bytes) to ensure ultra-low streaming latency
			chunk := make([]byte, 1024)
			n, err := reader.Read(chunk)
			if err != nil {
				// EOF or read error; process terminated
				log.Printf("[bimaxd] PTY master read EOF: %v", err)
				d.cancel()
				return
			}
			data := chunk[:n]

			// 1. Forward pristine, unaltered raw ANSI stream to mobile client
			select {
			case d.outCh <- data:
			default:
				// If channel is saturated, discard or handle backpressure
			}

			// 2. Feed side buffer for prompt pattern detection (never mutates outCh bytes)
			d.appendAndDetect(data)
		}
	}
}

// appendAndDetect manages the sliding window and executes non-destructive pattern matching
func (d *PTYDaemon) appendAndDetect(data []byte) {
	d.mu.Lock()
	defer d.mu.Unlock()

	d.buf.Write(data)

	// Trim sliding window from the front if buffer exceeds max size
	if d.buf.Len() > d.maxBufSize {
		overflow := d.buf.Len() - d.maxBufSize
		tmp := d.buf.Bytes()
		d.buf.Reset()
		d.buf.Write(tmp[overflow:])
	}

	// Strip ANSI escape sequences ONLY on a copy of the buffer
	plain := stripANSI(d.buf.Bytes())
	if promptText, ok := detectPrompt(plain); ok {
		select {
		case d.promptCh <- promptText:
		default:
			// Non-blocking drop if channel is full
		}
	}
}

// Write transmits input bytes (e.g. user keystrokes or "y\n") to child stdin
func (d *PTYDaemon) Write(b []byte) (int, error) {
	return d.ptyMaster.Write(b)
}

// Resize updates the PTY window size; kernel issues TIOCSWINSZ and fires SIGWINCH
func (d *PTYDaemon) Resize(rows, cols uint16) error {
	ws := &pty.Winsize{Rows: rows, Cols: cols}
	d.mu.Lock()
	defer d.mu.Unlock()
	return pty.Setsize(d.ptyMaster, ws)
}

// waitLoop monitors child process exit status
func (d *PTYDaemon) waitLoop() {
	err := d.cmd.Wait()
	if err != nil {
		log.Printf("[bimaxd] Child process exited with error: %v", err)
	}
	d.cancel()
}

// Close gracefully terminates the PTY session and releases descriptors
func (d *PTYDaemon) Close() error {
	d.cancel()
	_ = d.cmd.Process.Kill()
	return d.ptyMaster.Close()
}

// GetPGID returns the Process Group ID for the panic switch controller
func (d *PTYDaemon) GetPGID() int {
	return d.pgid
}

// Channel getters
func (d *PTYDaemon) Out() <-chan []byte    { return d.outCh }
func (d *PTYDaemon) Prompts() <-chan string { return d.promptCh }
```

### 3. Non-Destructive ANSI Stripping & Prompt Detection

Because escape sequences (such as cursor positioning, color resets, or bold font markers) can be interspersed throughout the raw byte stream, running regex directly on raw ANSI leads to missed matches. Stripping escape sequences on a **separate in-memory copy** solves this completely:

```go
package agent

import "regexp"

// Basic ANSI escape sequence stripper (CSI codes: ESC [ ... [A-Za-z])
var ansiRegexp = regexp.MustCompile(`\x1b\[[0-9;]*[A-Za-z]`)

func stripANSI(b []byte) []byte {
	return ansiRegexp.ReplaceAll(b, nil)
}

// Multi-pattern prompt matchers targeting interactive AI CLIs and shells
var promptPatterns = []*regexp.Regexp{
	regexp.MustCompile(`(?i)\[y/n\]`),
	regexp.MustCompile(`(?i)\(y/n\)`),
	regexp.MustCompile(`(?i)do you want to continue\?`),
	regexp.MustCompile(`(?i)do you want to proceed\?`),
	regexp.MustCompile(`(?i)allow this action\?`),
	regexp.MustCompile(`(?i)are you sure\?`),
}

// detectPrompt returns the detected prompt signature if matched
func detectPrompt(b []byte) (string, bool) {
	s := string(b)
	for _, re := range promptPatterns {
		if re.MatchString(s) {
			return re.String(), true
		}
	}
	return "", false
}
```

### 4. Dynamic Window Resizing & `SIGWINCH` Synchronization

#### A. Remote Mobile Client Window Resizing
When the mobile phone rotates orientation, opens the soft keyboard, or docks into **vivo / iQOO Office Kit**, the mobile client sends an E2EE control frame containing new `(rows, cols)` dimensions.

```go
// Invoked when an incoming WebSocket control packet arrives from Android
func OnClientResizeEvent(daemon *PTYDaemon, rows, cols int) {
	if err := daemon.Resize(uint16(rows), uint16(cols)); err != nil {
		log.Printf("[bimaxd] Failed to resize PTY: %v", err)
	} else {
		log.Printf("[bimaxd] Synchronized PTY size to %dx%d (SIGWINCH fired)", cols, rows)
	}
}
```

#### B. Local macOS Desktop Terminal Resizing (Development / Debug Mode)
When running `bimaxd` directly attached to a physical macOS terminal window, propagate desktop window resizes into the PTY:

```go
func AttachSIGWINCHResize(ptmx *os.File) {
	ch := make(chan os.Signal, 1)
	signal.Notify(ch, syscall.SIGWINCH)

	go func() {
		for range ch {
			// Copies size from current stdin to PTY master
			if err := pty.InheritSize(os.Stdin, ptmx); err != nil {
				log.Printf("[bimaxd] Error inheriting terminal size: %v", err)
			}
		}
	}()
	// Trigger initial size synchronization
	ch <- syscall.SIGWINCH
}
```
