### 1. Verification Summary & Principle of Least Privilege

Autonomous AI coding agents must **never** be provided with permanent, static personal access tokens (PATs) or long-lived cloud root credentials (`AWS_ACCESS_KEY_ID`, `AWS_SECRET_ACCESS_KEY`). If an agent is tricked by prompt injection in a dependency, or enters a destructive loop, long-lived credentials can lead to catastrophic cluster or repository compromise.

The **Short-Lived Scoped Token Broker** enforces dynamic least privilege by vending ephemeral tokens:
1. **GitHub App Installation Access Tokens (Repository-Scoped)**:
   - The Mac daemon signs an ephemeral RS256 JWT using the GitHub App's RSA private key ($9\text{ min}$ expiration).
   - Exchanges the JWT with GitHub API (`/app/installations/{id}/access_tokens`), scoping access strictly to specific repository IDs (`repository_ids: []int64`).
   - GitHub vends an installation token valid for **exactly 1 hour** with restricted permissions (`contents: write`, `pull_requests: write`).
2. **AWS STS AssumeRole (15-Minute Session Keys)**:
   - Invokes AWS Security Token Service (`sts:AssumeRole`) with `DurationSeconds: 900` (15 minutes) to mint temporary session keys (`AccessKeyId`, `SecretAccessKey`, `SessionToken`).
3. **In-Memory Per-Process Environment Injection**:
   - Vended tokens are injected **strictly into the child process environment memory** (`cmd.Env`).
   - Tokens are **never written to disk, `~/.zshrc`, `/etc/profile`, or git config**, vanishing completely from the OS when the process terminates.

```
+-----------------------------------------------------------------------------------------+
|                               SCOPED TOKEN BROKER PIPELINE                              |
|                                                                                         |
|  [Agent Spawns Git / Cloud Task]                                                        |
|           │                                                                             |
|           ▼                                                                             |
|  [bimaxd Scoped Token Broker (Go)]                                                      |
```

### 2. GitHub App Scoped Token Broker in Go (`github_broker.go`)

```go
package credbroker

import (
	"bytes"
	"context"
	"crypto/rsa"
	"encoding/json"
	"fmt"
	"net/http"
	"os"
	"time"

	"github.com/golang-jwt/jwt/v5"
)

// InstallationTokenResponse represents the response subset from GitHub API
type InstallationTokenResponse struct {
	Token        string    `json:"token"`
	ExpiresAt    time.Time `json:"expires_at"`
	Repositories []struct {
		ID   int64  `json:"id"`
		Name string `json:"name"`
	} `json:"repositories,omitempty"`
}

// LoadGitHubAppPrivateKey parses RSA private key from PEM file using golang-jwt parser
func LoadGitHubAppPrivateKey(pemPath string) (*rsa.PrivateKey, error) {
	pemBytes, err := os.ReadFile(pemPath)
	if err != nil {
		return nil, fmt.Errorf("read private key: %w", err)
	}
	key, err := jwt.ParseRSAPrivateKeyFromPEM(pemBytes)
	if err != nil {
		return nil, fmt.Errorf("parse RSA private key: %w", err)
	}
	return key, nil
}

// GenerateGitHubAppJWT generates an RS256 JWT valid for 9 minutes
func GenerateGitHubAppJWT(appID int64, key *rsa.PrivateKey) (string, error) {
	now := time.Now().UTC()

	claims := jwt.RegisteredClaims{
		Issuer:    fmt.Sprintf("%d", appID),                     // GitHub App ID as string
		IssuedAt:  jwt.NewNumericDate(now),                      // iat
		ExpiresAt: jwt.NewNumericDate(now.Add(9 * time.Minute)), // exp (< 10 min window)
	}

	token := jwt.NewWithClaims(jwt.SigningMethodRS256, claims)
	signed, err := token.SignedString(key)
	if err != nil {
		return "", fmt.Errorf("sign JWT: %w", err)
	}
	return signed, nil
}

// CreateInstallationAccessToken exchanges JWT for a 1-hour repository-scoped access token
func CreateInstallationAccessToken(
	ctx context.Context,
	appJWT string,
	installationID int64,
	repoIDs []int64,
) (*InstallationTokenResponse, error) {

	url := fmt.Sprintf("https://api.github.com/app/installations/%d/access_tokens", installationID)

	var body bytes.Buffer
	if len(repoIDs) > 0 {
		payload := struct {
			RepositoryIDs []int64 `json:"repository_ids"`
		}{
			RepositoryIDs: repoIDs,
		}
		if err := json.NewEncoder(&body).Encode(payload); err != nil {
			return nil, fmt.Errorf("encode payload: %w", err)
		}
	}

	req, err := http.NewRequestWithContext(ctx, http.MethodPost, url, &body)
	if err != nil {
		return nil, fmt.Errorf("new request: %w", err)
	}

	req.Header.Set("Authorization", "Bearer "+appJWT)
	req.Header.Set("Accept", "application/vnd.github+json")
	req.Header.Set("X-GitHub-Api-Version", "2022-11-28")

	client := &http.Client{Timeout: 10 * time.Second}
	resp, err := client.Do(req)
	if err != nil {
		return nil, fmt.Errorf("POST installation token: %w", err)
	}
	defer resp.Body.Close()

	if resp.StatusCode != http.StatusCreated {
		var errBody bytes.Buffer
		_, _ = errBody.ReadFrom(resp.Body)
		return nil, fmt.Errorf("unexpected status %s: %s", resp.Status, errBody.String())
	}

	var tokenResp InstallationTokenResponse
	if err := json.NewDecoder(resp.Body).Decode(&tokenResp); err != nil {
		return nil, fmt.Errorf("decode response: %w", err)
	}

	return &tokenResp, nil
}

// MintGitHubTokenForAgent is the high-level broker entrypoint
func MintGitHubTokenForAgent(ctx context.Context, appID, installationID int64, repoIDs []int64, pemPath string) (string, time.Time, error) {
	key, err := LoadGitHubAppPrivateKey(pemPath)
	if err != nil {
		return "", time.Time{}, err
	}

	jwtToken, err := GenerateGitHubAppJWT(appID, key)
	if err != nil {
		return "", time.Time{}, err
	}

	instTok, err := CreateInstallationAccessToken(ctx, jwtToken, installationID, repoIDs)
	if err != nil {
		return "", time.Time{}, err
	}

	return instTok.Token, instTok.ExpiresAt, nil
}
```

### 3. AWS STS Ephemeral AssumeRole Broker (`aws_broker.go`)

```go
package credbroker

import (
	"context"
	"fmt"
	"time"

	"github.com/aws/aws-sdk-go-v2/aws"
	"github.com/aws/aws-sdk-go-v2/config"
	"github.com/aws/aws-sdk-go-v2/service/sts"
)

type TemporaryAWSCreds struct {
	AccessKeyID     string    `json:"access_key_id"`
	SecretAccessKey string    `json:"secret_access_key"`
	SessionToken    string    `json:"session_token"`
	Expiration      time.Time `json:"expiration"`
}

// MintAWSTemporaryCreds uses sts:AssumeRole to return 15-minute temporary credentials
func MintAWSTemporaryCreds(ctx context.Context, roleARN, sessionName string) (*TemporaryAWSCreds, error) {
	cfg, err := config.LoadDefaultConfig(ctx)
	if err != nil {
		return nil, fmt.Errorf("load AWS config: %w", err)
	}

	stsClient := sts.NewFromConfig(cfg)

	input := &sts.AssumeRoleInput{
		RoleArn:         aws.String(roleARN),
		RoleSessionName: aws.String(sessionName),
		DurationSeconds: aws.Int32(900), // 15 minutes minimum duration
	}

	out, err := stsClient.AssumeRole(ctx, input)
	if err != nil {
		return nil, fmt.Errorf("assume role: %w", err)
	}

	creds := out.Credentials
	if creds == nil {
		return nil, fmt.Errorf("no credentials in AssumeRole output")
	}

	return &TemporaryAWSCreds{
		AccessKeyID:     aws.ToString(creds.AccessKeyId),
		SecretAccessKey: aws.ToString(creds.SecretAccessKey),
		SessionToken:    aws.ToString(creds.SessionToken),
		Expiration:      aws.ToTime(creds.Expiration),
	}, nil
}
```

### 4. In-Memory Per-Process Environment Injection (`process_spawner.go`)

```go
package credbroker

import (
	"os"
	"os/exec"
	"strings"
	"syscall"
)

// AgentCreds bundles the ephemeral secrets injected into child memory
type AgentCreds struct {
	GitHubToken        string
	GitHubTokenExpiry  string // ISO8601 UTC string
	AWSAccessKeyID     string
	AWSSecretAccessKey string
	AWSSessionToken    string
}

// Blocked master environment variables stripped from child processes
var blacklistedMasterEnvs = []string{
	"GITHUB_TOKEN",
	"GH_TOKEN",
	"AWS_ACCESS_KEY_ID",
	"AWS_SECRET_ACCESS_KEY",
	"AWS_SESSION_TOKEN",
	"CLOUDFLARE_API_KEY",
	"ANTHROPIC_API_KEY",
	"OPENAI_API_KEY",
}

// SpawnAgentWithCreds starts the agent child process with injected ephemeral credentials
func SpawnAgentWithCreds(agentPath string, args []string, workingDir string, creds AgentCreds) (*exec.Cmd, error) {
	cmd := exec.Command(agentPath, args...)
	if workingDir != "" {
		cmd.Dir = workingDir
	}

	// Make child leader of a dedicated process group for atomic signal management
	cmd.SysProcAttr = &syscall.SysProcAttr{
		Setpgid: true,
	}

	// Filter out static master credentials from the host daemon's environment
	baseEnv := os.Environ()
	cleanEnv := make([]string, 0, len(baseEnv)+5)

	for _, item := range baseEnv {
		isBlocked := false
		for _, blocked := range blacklistedMasterEnvs {
			if strings.HasPrefix(item, blocked+"=") {
				isBlocked = true
				break
			}
		}
		if !isBlocked {
			cleanEnv = append(cleanEnv, item)
		}
	}

	// Inject brokered ephemeral credentials strictly into child cmd.Env
	cleanEnv = append(cleanEnv,
		"GITHUB_TOKEN="+creds.GitHubToken,
		"GITHUB_TOKEN_EXPIRES_AT="+creds.GitHubTokenExpiry,
		"AWS_ACCESS_KEY_ID="+creds.AWSAccessKeyID,
		"AWS_SECRET_ACCESS_KEY="+creds.AWSSecretAccessKey,
		"AWS_SESSION_TOKEN="+creds.AWSSessionToken,
	)

	cmd.Env = cleanEnv
	cmd.Stdout = os.Stdout
	cmd.Stderr = os.Stderr

	// Start child process without mutating host OS environment
	if err := cmd.Start(); err != nil {
		return nil, err
	}

	return cmd, nil
}
```
