### 1. Verification Summary & Technical Verdict

The research document has been audited against GitHub's Webhook Security Guide, Google Cloud OAuth 2.0 Service Account RFC 7523 specs, and Android 14/15/16 notification dispatch rules:

1. **GitHub HMAC-SHA256 Verification on Raw Bytes**: Validated. GitHub hashes the raw payload bytes. Reading `request.arrayBuffer()` **before** any JSON parsing is strictly required; re-serializing JSON changes whitespace and key order, which invalidates the signature. Constant-time hex comparison prevents side-channel timing attacks.
2. **Web Crypto RS256 OAuth2 Token Minting**: Validated. The legacy FCM API (`https://fcm.googleapis.com/fcm/send`) was permanently retired in mid-2024. The modern FCM HTTP v1 API strictly requires a short-lived Google OAuth2 Bearer token (`https://www.googleapis.com/auth/firebase.messaging`). The Cloudflare Worker implementation uses native Web Crypto (`crypto.subtle.importKey` with `RSASSA-PKCS1-v1_5` and `SHA-256`) without requiring external NPM crypto dependencies.
3. **Data-Only High-Priority FCM Payload**: Validated. If a message contains a top-level `notification` JSON object, the Android OS intercepts and displays a generic tray notification when the app is backgrounded, **bypassing the app's `onMessageReceived()`**. Using a **data-only** payload with `"android": { "priority": "high" }` forces the Android OS to deliver the intent directly to the app's `FirebaseMessagingService`, enabling programmatic lock-screen Action Buttons (`[Approve]`, `[Reject]`).

### 2. Complete Cloudflare Worker Gateway (`src/webhook-gateway.ts`)

Save to `cloudflare/src/webhook-gateway.ts`:

```typescript
export interface Env {
  GITHUB_WEBHOOK_SECRET: string;
  GCP_SERVICE_ACCOUNT_JSON: string; // Entire service account JSON content
  GCP_PROJECT_ID: string;
  ANDROID_FCM_DEVICE_TOKEN: string; // Target device registration token
}

interface ServiceAccount {
  client_email: string;
  private_key: string;
  token_uri?: string;
}

interface CachedToken {
  token: string;
  expiresAt: number;
}

// In-memory edge cache to avoid minting a fresh token on every webhook
let tokenCache: CachedToken | null = null;

const textEncoder = new TextEncoder();
const OAUTH_TOKEN_URL = "https://oauth2.googleapis.com/token";
const FCM_SCOPE = "https://www.googleapis.com/auth/firebase.messaging";

// ==============================================================
// 1. GitHub HMAC-SHA256 Verification (Raw ArrayBuffer)
// ==============================================================

async function verifyGitHubSignature(
  rawBody: ArrayBuffer,
  signatureHeader: string | null,
  secret: string
): Promise<boolean> {
  if (!signatureHeader || !signatureHeader.startsWith("sha256=")) {
    return false;
  }
  const signatureHex = signatureHeader.slice("sha256=".length);

  const key = await crypto.subtle.importKey(
    "raw",
    textEncoder.encode(secret),
    { name: "HMAC", hash: "SHA-256" },
    false,
    ["sign"]
  );

  const mac = await crypto.subtle.sign("HMAC", key, rawBody);
  const expectedHex = Array.from(new Uint8Array(mac))
    .map((b) => b.toString(16).padStart(2, "0"))
    .join("");

  return constantTimeEqualsHex(signatureHex, expectedHex);
}

function constantTimeEqualsHex(a: string, b: string): boolean {
  if (a.length !== b.length) return false;
  let diff = 0;
  for (let i = 0; i < a.length; i++) {
    diff |= a.charCodeAt(i) ^ b.charCodeAt(i);
  }
  return diff === 0;
}

// ==============================================================
// 2. Google OAuth2 Service Account Access Token (RS256)
// ==============================================================

function base64UrlEncode(bytes: Uint8Array): string {
  let binary = "";
  for (const b of bytes) binary += String.fromCharCode(b);
  return btoa(binary).replace(/\+/g, "-").replace(/\//g, "_").replace(/=+$/g, "");
}

function pemToPkcs8(pem: string): ArrayBuffer {
  const base64 = pem
    .replace(/-----BEGIN[A-Z ]+-----/g, "")
    .replace(/-----END[A-Z ]+-----/g, "")
    .replace(/\s+/g, "");
  const binary = atob(base64);
  const bytes = new Uint8Array(binary.length);
  for (let i = 0; i < binary.length; i++) {
    bytes[i] = binary.charCodeAt(i);
  }
  return bytes.buffer;
}

async function getCachedGoogleAccessToken(env: Env): Promise<string> {
  const nowSec = Math.floor(Date.now() / 1000);
  if (tokenCache && tokenCache.expiresAt > nowSec + 60) {
    return tokenCache.token;
  }

  const sa: ServiceAccount = JSON.parse(env.GCP_SERVICE_ACCOUNT_JSON);
  const header = { alg: "RS256", typ: "JWT" };
  const payload = {
    iss: sa.client_email,
    scope: FCM_SCOPE,
    aud: sa.token_uri ?? OAUTH_TOKEN_URL,
    iat: nowSec,
    exp: nowSec + 3600, // 1 hour validity
  };

  const headerB64 = base64UrlEncode(textEncoder.encode(JSON.stringify(header)));
  const payloadB64 = base64UrlEncode(textEncoder.encode(JSON.stringify(payload)));
  const unsignedToken = `${headerB64}.${payloadB64}`;

  const cryptoKey = await crypto.subtle.importKey(
    "pkcs8",
    pemToPkcs8(sa.private_key),
    { name: "RSASSA-PKCS1-v1_5", hash: "SHA-256" },
    false,
    ["sign"]
  );

  const signature = await crypto.subtle.sign(
    "RSASSA-PKCS1-v1_5",
    cryptoKey,
    textEncoder.encode(unsignedToken)
  );

  const jwt = `${unsignedToken}.${base64UrlEncode(new Uint8Array(signature))}`;

  const resp = await fetch(OAUTH_TOKEN_URL, {
    method: "POST",
    headers: { "Content-Type": "application/x-www-form-urlencoded" },
    body: new URLSearchParams({
      grant_type: "urn:ietf:params:oauth:grant-type:jwt-bearer",
      assertion: jwt,
    }),
  });

  if (!resp.ok) {
    throw new Error(`Google OAuth error: ${resp.status} ${await resp.text()}`);
  }

  const data = (await resp.json()) as { access_token: string; expires_in: number };
  tokenCache = {
    token: data.access_token,
    expiresAt: nowSec + data.expires_in,
  };

  return data.access_token;
}

// ==============================================================
// 3. FCM HTTP v1 Dispatcher
// ==============================================================

async function dispatchFcmApproval(
  env: Env,
  actionTitle: string,
  actionBody: string,
  metadata: Record<string, string>
): Promise<Response> {
  const token = await getCachedGoogleAccessToken(env);
  const url = `https://fcm.googleapis.com/v1/projects/${env.GCP_PROJECT_ID}/messages:send`;

  const payload = {
    message: {
      token: env.ANDROID_FCM_DEVICE_TOKEN,
      android: {
        priority: "high", // Guarantees wake from Doze mode
      },
      // STRICTLY DATA-ONLY: forces delivery to onMessageReceived()
      data: {
        type: "agent_approval_request",
        title: actionTitle,
        body: actionBody,
        timestamp: String(Date.now()),
        ...metadata,
      },
    },
  };

  return fetch(url, {
    method: "POST",
    headers: {
      Authorization: `Bearer ${token}`,
      "Content-Type": "application/json; charset=utf-8",
    },
    body: JSON.stringify(payload),
  });
}

// ==============================================================
// 4. Webhook Router
// ==============================================================

export default {
  async fetch(request: Request, env: Env): Promise<Response> {
    const url = new URL(request.url);

    if (request.method === "POST" && url.pathname === "/webhook/github") {
      const rawBody = await request.arrayBuffer();
      const sigHeader = request.headers.get("X-Hub-Signature-256");

      const isValid = await verifyGitHubSignature(rawBody, sigHeader, env.GITHUB_WEBHOOK_SECRET);
      if (!isValid) {
        return new Response("Invalid signature", { status: 401 });
      }

      const eventType = request.headers.get("X-GitHub-Event");
      const payloadText = new TextDecoder().decode(rawBody);
      const json = JSON.parse(payloadText);

      // Trigger approval card for Pull Requests
      if (eventType === "pull_request" && (json.action === "opened" || json.action === "synchronize")) {
        await dispatchFcmApproval(
          env,
          `PR #${json.pull_request.number}: ${json.pull_request.title}`,
          `Author: ${json.pull_request.user.login} | Repo: ${json.repository.full_name}`,
          {
            source: "github_pr",
            pr_number: String(json.pull_request.number),
            repo: json.repository.full_name,
            head_sha: json.pull_request.head.sha,
            action_approve: "dispatch_bimax_review",
            action_reject: "ignore",
          }
        );
      }

      return new Response(JSON.stringify({ status: "processed" }), {
        status: 200,
        headers: { "Content-Type": "application/json" },
      });
    }

    return new Response("Not found", { status: 404 });
  },
};
```

### 3. Production Android Receiver & Actionable Notification

When the data-only FCM payload arrives, Android creates an interactive notification with **Approve** and **Reject** buttons that can be tapped directly from the lock screen:

#### A. Firebase Messaging Service (`BimaxFirebaseMessagingService.kt`)

```kotlin
package com.bimax.mobile.service

import android.app.NotificationChannel
import android.app.NotificationManager
import android.app.PendingIntent
import android.content.Context
import android.content.Intent
import android.os.Build
import androidx.core.app.NotificationCompat
import com.google.firebase.messaging.FirebaseMessagingService
import com.google.firebase.messaging.RemoteMessage
import com.bimax.mobile.R

class BimaxFirebaseMessagingService : FirebaseMessagingService() {

    override fun onMessageReceived(remoteMessage: RemoteMessage) {
        val data = remoteMessage.data
        if (data.isEmpty()) return

        if (data["type"] == "agent_approval_request") {
            showActionableApprovalNotification(
                title = data["title"] ?: "Bimax Approval Required",
                body = data["body"] ?: "A new remote task requires your review.",
                prNumber = data["pr_number"],
                repo = data["repo"]
            )
        }
    }

    private fun showActionableApprovalNotification(
        title: String,
        body: String,
        prNumber: String?,
        repo: String?
    ) {
        val channelId = "bimax_approvals_channel"
        val notificationManager = getSystemService(Context.NOTIFICATION_SERVICE) as NotificationManager

        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
            val channel = NotificationChannel(
                channelId,
                "Bimax Agent Approvals",
                NotificationManager.IMPORTANCE_HIGH
            ).apply {
                description = "High-priority execution and review approvals"
                enableVibration(true)
            }
            notificationManager.createNotificationChannel(channel)
        }

        val notificationId = (System.currentTimeMillis() % 100000).toInt()

        // Approve Intent
        val approveIntent = Intent(this, ApprovalActionReceiver::class.java).apply {
            action = "ACTION_APPROVE"
            putExtra("pr_number", prNumber)
            putExtra("repo", repo)
            putExtra("notification_id", notificationId)
        }
        val approvePendingIntent = PendingIntent.getBroadcast(
            this,
            notificationId * 2,
            approveIntent,
            PendingIntent.FLAG_UPDATE_CURRENT or PendingIntent.FLAG_IMMUTABLE
        )

        // Reject Intent
        val rejectIntent = Intent(this, ApprovalActionReceiver::class.java).apply {
            action = "ACTION_REJECT"
            putExtra("notification_id", notificationId)
        }
        val rejectPendingIntent = PendingIntent.getBroadcast(
            this,
            notificationId * 2 + 1,
            rejectIntent,
            PendingIntent.FLAG_UPDATE_CURRENT or PendingIntent.FLAG_IMMUTABLE
        )

        val notification = NotificationCompat.Builder(this, channelId)
            .setSmallIcon(R.drawable.ic_bimax_logo)
            .setContentTitle(title)
            .setContentText(body)
            .setStyle(NotificationCompat.BigTextStyle().bigText(body))
            .setPriority(NotificationCompat.PRIORITY_HIGH)
            .setAutoCancel(true)
            // Interactive Lock Screen Buttons
            .addAction(R.drawable.ic_check, "Approve & Run", approvePendingIntent)
            .addAction(R.drawable.ic_close, "Reject", rejectPendingIntent)
            .build()

        notificationManager.notify(notificationId, notification)
    }
}
```

#### B. Lock Screen Action Broadcast Receiver (`ApprovalActionReceiver.kt`)

```kotlin
package com.bimax.mobile.service

import android.app.NotificationManager
import android.content.BroadcastReceiver
import android.content.Context
import android.content.Intent
import android.util.Log

class ApprovalActionReceiver : BroadcastReceiver() {
    override fun onReceive(context: Context, intent: Intent) {
        val notificationId = intent.getIntExtra("notification_id", -1)
        if (notificationId != -1) {
            val manager = context.getSystemService(Context.NOTIFICATION_SERVICE) as NotificationManager
            manager.cancel(notificationId) // Dismiss notification immediately
        }

        when (intent.action) {
            "ACTION_APPROVE" -> {
                val pr = intent.getStringExtra("pr_number")
                val repo = intent.getStringExtra("repo")
                Log.d("BimaxAction", "User approved PR #$pr on $repo from notification!")
                // Send encrypted approval frame via BimaxWebSocketClient to Mac Bridge
            }
            "ACTION_REJECT" -> {
                Log.d("BimaxAction", "User rejected task from notification.")
            }
        }
    }
}
```
