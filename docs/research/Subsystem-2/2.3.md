### 1. Verification Summary & Hardware Security Architecture

Android devices possess dedicated hardware security processors isolated from the main Android OS kernel:
1. **Trusted Execution Environment (TEE)**: An ARM TrustZone-isolated execution environment running a secure microkernel OS.
2. **StrongBox Keymaster**: A physically dedicated discrete hardware chip (Secure Element / HSM with physical tamper resistance, side-channel attack countermeasures, and true random number generators).
3. **NIST EC P-256 vs. Curve25519 Clarification**:
   - **Critical Engineering Insight**: `AndroidKeyStore` **does not support Curve25519 / Ed25519 for hardware-backed signing** on standard Android APIs. Requesting Curve25519 inside `KeyGenParameterSpec` throws an `InvalidAlgorithmParameterException`.
   - **The Dual-Key Strategy**:
     - **Session Transport (Layer 2)**: Uses **Curve25519 (X25519)** in user-space via `noise-java` for the fast streaming Noise_IK tunnel.
     - **Command Approval Authorization (Layer 3)**: Uses **NIST EC P-256 (`secp256r1`) + `SHA256withECDSA`** inside the hardware-backed **Android Keystore (StrongBox/TEE)**.
4. **Permanent Invalidation on Biometric Changes (`setInvalidatedByBiometricEnrollment(true)`)**:
   - If an attacker learns the phone's lock-screen PIN and enrolls a new fingerprint, the Keystore permanently and irreversibly destroys the private key. Subsequent calls to `Signature.initSign` throw `KeyPermanentlyInvalidatedException`.
5. **Auth-Per-Use (`setUserAuthenticationParameters(0, ...)`):**
   - Setting timeout to $0$ seconds guarantees that authorization is **valid for only a single cryptographic signature**. No background process can reuse an authorization window.

### 2. StrongBox Keypair Generator with TEE Fallback (`KeystoreManager.kt`)

```kotlin
package com.bimax.mobile.security

import android.content.Context
import android.content.pm.PackageManager
import android.os.Build
import android.security.keystore.KeyGenParameterSpec
import android.security.keystore.KeyInfo
import android.security.keystore.KeyProperties
import android.security.keystore.StrongBoxUnavailableException
import java.security.KeyFactory
import java.security.KeyPair
import java.security.KeyPairGenerator
import java.security.KeyStore
import java.security.spec.ECGenParameterSpec

private const val ANDROID_KEYSTORE = "AndroidKeyStore"
private const val SIGNING_KEY_ALIAS = "bimax_destructive_signing_key"

class KeystoreManager(private val context: Context) {

    private val keyStore = KeyStore.getInstance(ANDROID_KEYSTORE).apply { load(null) }

    fun getOrCreateSigningKeyPair(): KeyPair {
        val existingEntry = keyStore.getEntry(SIGNING_KEY_ALIAS, null) as? KeyStore.PrivateKeyEntry
        if (existingEntry != null) {
            return KeyPair(existingEntry.certificate.publicKey, existingEntry.privateKey)
        }

        val kpg = KeyPairGenerator.getInstance(
            KeyProperties.KEY_ALGORITHM_EC,
            ANDROID_KEYSTORE
        )

        fun buildSpec(isStrongBox: Boolean): KeyGenParameterSpec {
            val builder = KeyGenParameterSpec.Builder(
                SIGNING_KEY_ALIAS,
                KeyProperties.PURPOSE_SIGN or KeyProperties.PURPOSE_VERIFY
            )
                .setAlgorithmParameterSpec(ECGenParameterSpec("secp256r1")) // NIST P-256
                .setDigests(KeyProperties.DIGEST_SHA256)
                .setUserAuthenticationRequired(true)
                // 0 timeout = Authentication required for EVERY individual signature operation
                .setUserAuthenticationParameters(
                    0,
                    KeyProperties.AUTH_BIOMETRIC_STRONG or KeyProperties.AUTH_DEVICE_CREDENTIAL
                )
                // Irreversibly destroy key if a new fingerprint is enrolled
                .setInvalidatedByBiometricEnrollment(true)

            if (isStrongBox && Build.VERSION.SDK_INT >= Build.VERSION_CODES.P) {
                builder.setIsStrongBoxBacked(true)
            }
            return builder.build()
        }

        val hasStrongBox = context.packageManager.hasSystemFeature(PackageManager.FEATURE_STRONGBOX_KEYSTORE)

        return try {
            // Priority 1: Hardware StrongBox (Dedicated HSM chip)
            kpg.initialize(buildSpec(isStrongBox = hasStrongBox))
            kpg.generateKeyPair()
        } catch (e: StrongBoxUnavailableException) {
            // Priority 2: Standard TEE (ARM TrustZone)
            kpg.initialize(buildSpec(isStrongBox = false))
            kpg.generateKeyPair()
        }
    }

    // Verify key resides in secure hardware
    fun isInsideSecureHardware(): Boolean {
        val entry = keyStore.getEntry(SIGNING_KEY_ALIAS, null) as? KeyStore.PrivateKeyEntry ?: return false
        val factory = KeyFactory.getInstance(entry.privateKey.algorithm, ANDROID_KEYSTORE)
        val keyInfo = factory.getKeySpec(entry.privateKey, KeyInfo::class.java) as KeyInfo
        return keyInfo.isInsideSecureHardware
    }

    // Wipe key if permanently invalidated
    fun deleteKey() {
        if (keyStore.containsAlias(SIGNING_KEY_ALIAS)) {
            keyStore.deleteEntry(SIGNING_KEY_ALIAS)
        }
    }
}
```

### 3. BiometricPrompt + CryptoObject Authorizer (`CommandAuthorizer.kt`)

```kotlin
package com.bimax.mobile.security

import android.security.keystore.KeyPermanentlyInvalidatedException
import android.security.keystore.UserNotAuthenticatedException
import androidx.biometric.BiometricManager
import androidx.biometric.BiometricPrompt
import androidx.core.content.ContextCompat
import androidx.fragment.app.FragmentActivity
import java.nio.ByteBuffer
import java.nio.ByteOrder
import java.security.Signature
import java.util.concurrent.Executor

sealed class DestructiveAuthError(message: String) : Exception(message) {
    class UserCanceled(val code: Int, msg: String) : DestructiveAuthError(msg)
    class Lockout(val code: Int, msg: String) : DestructiveAuthError(msg)
    class KeyInvalidated(msg: String) : DestructiveAuthError(msg)
    class HardwareUnavailable(val code: Int, msg: String) : DestructiveAuthError(msg)
    class Fatal(val code: Int, msg: String) : DestructiveAuthError(msg)
}

class CommandAuthorizer(
    private val activity: FragmentActivity,
    private val keystoreManager: KeystoreManager,
    private val executor: Executor = ContextCompat.getMainExecutor(activity)
) {
    private var pendingPayload: ByteArray? = null
    private var pendingCallback: ((Result<ByteArray>) -> Unit)? = null

    private val biometricPrompt = BiometricPrompt(
        activity,
        executor,
        object : BiometricPrompt.AuthenticationCallback() {
            override fun onAuthenticationSucceeded(result: BiometricPrompt.AuthenticationResult) {
                val signature = result.cryptoObject?.signature
                val payload = pendingPayload

                if (signature == null || payload == null) {
                    pendingCallback?.invoke(Result.failure(IllegalStateException("CryptoObject or payload missing")))
                    return
                }

                try {
                    signature.update(payload)
                    val signatureBytes = signature.sign()
                    pendingCallback?.invoke(Result.success(signatureBytes))
                } catch (e: KeyPermanentlyInvalidatedException) {
                    keystoreManager.deleteKey() // Wipe compromised key alias
                    pendingCallback?.invoke(Result.failure(DestructiveAuthError.KeyInvalidated("Biometric enrollment changed. Re-pairing required.")))
                } catch (e: Exception) {
                    pendingCallback?.invoke(Result.failure(e))
                } finally {
                    cleanup()
                }
            }

            override fun onAuthenticationError(errorCode: Int, errString: CharSequence) {
                val error = when (errorCode) {
                    BiometricPrompt.ERROR_NEGATIVE_BUTTON,
                    BiometricPrompt.ERROR_USER_CANCELED,
                    BiometricPrompt.ERROR_CANCELED -> {
                        // User dismissed or clicked Cancel -> Clean non-error abort
                        DestructiveAuthError.UserCanceled(errorCode, errString.toString())
                    }
                    BiometricPrompt.ERROR_LOCKOUT,
                    BiometricPrompt.ERROR_LOCKOUT_PERMANENT -> {
                        DestructiveAuthError.Lockout(errorCode, errString.toString())
                    }
                    BiometricPrompt.ERROR_NO_BIOMETRICS,
                    BiometricPrompt.ERROR_HW_NOT_PRESENT,
                    BiometricPrompt.ERROR_HW_UNAVAILABLE -> {
                        DestructiveAuthError.HardwareUnavailable(errorCode, errString.toString())
                    }
                    else -> {
                        DestructiveAuthError.Fatal(errorCode, errString.toString())
                    }
                }
                pendingCallback?.invoke(Result.failure(error))
                cleanup()
            }

            override fun onAuthenticationFailed() {
                // Intermediate fingerprint mismatch; system prompt stays visible for retry
            }
        }
    )

    fun authorizeAndSign(
        commandHash: ByteArray, // 32-byte SHA-256 of proposed command
        timestampMillis: Long,  // Epoch millis
        nonce: ByteArray,       // 16-byte single-use nonce
        callback: (Result<ByteArray>) -> Unit
    ) {
        val biometricManager = BiometricManager.from(activity)
        val canAuth = biometricManager.canAuthenticate(
            BiometricManager.Authenticators.BIOMETRIC_STRONG or BiometricManager.Authenticators.DEVICE_CREDENTIAL
        )
        if (canAuth != BiometricManager.BIOMETRIC_SUCCESS) {
            callback(Result.failure(IllegalStateException("Biometric hardware not ready or un-enrolled")))
            return
        }

        // Build binary approval payload: [commandHash (32B) | timestamp (8B) | nonce (16B)]
        pendingPayload = ByteBuffer.allocate(commandHash.size + 8 + nonce.size)
            .order(ByteOrder.BIG_ENDIAN)
            .put(commandHash)
            .putLong(timestampMillis)
            .put(nonce)
            .array()

        pendingCallback = callback

        val keyPair = keystoreManager.getOrCreateSigningKeyPair()
        val signature = Signature.getInstance("SHA256withECDSA")

        try {
            signature.initSign(keyPair.private)
        } catch (e: KeyPermanentlyInvalidatedException) {
            keystoreManager.deleteKey()
            callback(Result.failure(DestructiveAuthError.KeyInvalidated("Biometric enrollment changed. Re-pairing required.")))
            cleanup()
            return
        }

        val cryptoObject = BiometricPrompt.CryptoObject(signature)

        val promptInfo = BiometricPrompt.PromptInfo.Builder()
            .setTitle("Authorize Sensitive Action")
            .setSubtitle("Bimax Mac Agent requires your hardware authorization")
            .setDescription("Command execution requires unforgeable biometric signature.")
            .setAllowedAuthenticators(
                BiometricManager.Authenticators.BIOMETRIC_STRONG or BiometricManager.Authenticators.DEVICE_CREDENTIAL
            )
            .setNegativeButtonText("Deny Execution")
            .build()

        biometricPrompt.authenticate(promptInfo, cryptoObject)
    }

    private fun cleanup() {
        pendingPayload = null
        pendingCallback = null
    }
}
```

### 4. Mac Daemon Verification (Go)

The Mac host verifies the signature using the paired Android EC public key:

```go
package crypto

import (
	"crypto/ecdsa"
	"crypto/sha256"
	"encoding/asn1"
	"fmt"
	"math/big"
)

type ECDSASignature struct {
	R, S *big.Int
}

// VerifyCommandApproval verifies the hardware signature over [commandHash | timestampMillis | nonce]
func VerifyCommandApproval(androidECPubKey *ecdsa.PublicKey, payloadBytes []byte, signatureASN1 []byte) bool {
	var sig ECDSASignature
	_, err := asn1.Unmarshal(signatureASN1, &sig)
	if err != nil {
		return false
	}

	hash := sha256.Sum256(payloadBytes)
	return ecdsa.Verify(androidECPubKey, hash[:], sig.R, sig.S)
}
```
