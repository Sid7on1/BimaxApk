### 1. Verification Summary & Dual-Mode Ergonomics

Developers interact with autonomous agents in two fundamentally distinct environments:
1. **On-the-Go (Compact Portrait, $<600\text{dp}$)**: Single-column, bottom-navigation UI optimized for one-handed thumb reachability, quick voice commands, and swipeable approval cards.
2. **At a Desk / Monitor (Expanded Landscape, $\ge 840\text{dp}$)**: Connected via USB-C monitor, wireless screen cast, or **vivo / iQOO Office Kit / Samsung DeX**. 

By utilizing the modern **Material 3 Adaptive Suite** (`androidx.compose.material3.adaptive`):
* The layout dynamically adapts **without restarting the Activity** when docked or cast to a monitor.
* **`NavigationSuiteScaffold`** automatically adapts navigation: rendering a `NavigationBar` on compact mobile, a `NavigationRail` on medium foldables, and a desktop navigation rail or drawer on expanded external displays.
* In `Expanded` mode ($\ge 840\text{dp}$ width), **`SupportingPaneScaffold`** renders an IDE 3-Pane Cockpit:
  - **Left Pane (`mainPane`)**: Active tasks, subagent hierarchy tree, and connector feeds.
  - **Center Pane (`supportingPane`)**: Monospace ANSI terminal and side-by-side Git diff inspector.
  - **Right Pane (`extraPane`)**: Real-time Mac vitals (CPU/RAM/thermals), browser preview, and token spend telemetry.
* Hardware keyboard shortcuts (`Ctrl + Enter`, `Ctrl + \``) and mouse wheel gestures are intercepted at the root scaffold level.

### 3. Complete Adaptive Shell Implementation (`OfficeKitAdaptiveApp.kt`)

```kotlin
package com.bimax.mobile.ui.desktop

import androidx.compose.foundation.background
import androidx.compose.foundation.layout.*
import androidx.compose.material.icons.Icons
import androidx.compose.material.icons.filled.Chat
import androidx.compose.material.icons.filled.Checklist
import androidx.compose.material.icons.filled.Dashboard
import androidx.compose.material.icons.filled.Terminal
import androidx.compose.material3.*
import androidx.compose.material3.adaptive.ExperimentalMaterial3AdaptiveApi
import androidx.compose.material3.adaptive.WindowAdaptiveInfo
import androidx.compose.material3.adaptive.currentWindowAdaptiveInfo
import androidx.compose.material3.adaptive.layout.ExperimentalMaterial3AdaptiveLayoutApi
import androidx.compose.material3.adaptive.layout.SupportingPaneScaffold
import androidx.compose.material3.adaptive.navigation.suite.NavigationSuite
import androidx.compose.material3.adaptive.navigation.suite.NavigationSuiteScaffold
import androidx.compose.material3.adaptive.navigation.suite.NavigationSuiteScaffoldDefaults
import androidx.compose.runtime.*
import androidx.compose.ui.Modifier
import androidx.compose.ui.graphics.Color
import androidx.compose.ui.graphics.vector.ImageVector
import androidx.compose.ui.input.key.*
import androidx.compose.ui.unit.dp
import androidx.window.core.layout.WindowWidthSizeClass
import com.bimax.mobile.ui.terminal.TerminalBuffer
import com.bimax.mobile.ui.terminal.TerminalScreen

enum class OfficeDestination(val label: String, val icon: ImageVector) {
    COCKPIT("Cockpit", Icons.Default.Dashboard),
    CHAT("Chat", Icons.Default.Chat),
    TERMINAL("Terminal", Icons.Default.Terminal),
    APPROVALS("Approvals", Icons.Default.Checklist)
}

@OptIn(ExperimentalMaterial3AdaptiveApi::class)
@Composable
fun OfficeKitAdaptiveApp(
    terminalBuffer: TerminalBuffer,
    onSendKey: (ByteArray) -> Unit,
    onSendPrompt: () -> Unit,
    onToggleTerminal: () -> Unit
) {
    val adaptiveInfo = currentWindowAdaptiveInfo()
    val widthClass = adaptiveInfo.windowSizeClass.windowWidthSizeClass

    Box(
        modifier = Modifier
            .fillMaxSize()
            .onPreviewKeyEvent { event ->
                // Global physical keyboard shortcut handler for Desktop / Office Kit mode
                if (event.type == KeyEventType.KeyUp) {
                    val ctrl = event.isCtrlPressed
                    when (event.key) {
                        Key.Enter -> {
                            if (ctrl) {
                                onSendPrompt()
                                true
                            } else false
                        }
                        Key.Grave -> { // Backquote / Tilde (`)
                            if (ctrl) {
                                onToggleTerminal()
                                true
                            } else false
                        }
                        else -> false
                    }
                } else false
            }
    ) {
        when (widthClass) {
            WindowWidthSizeClass.COMPACT -> CompactOfficeShell(adaptiveInfo, terminalBuffer, onSendKey)
            WindowWidthSizeClass.MEDIUM -> MediumOfficeShell(adaptiveInfo, terminalBuffer, onSendKey)
            WindowWidthSizeClass.EXPANDED -> ExpandedOfficeShell(adaptiveInfo, terminalBuffer, onSendKey)
            else -> ExpandedOfficeShell(adaptiveInfo, terminalBuffer, onSendKey)
        }
    }
}

// 1. Compact: Phone portrait with bottom NavigationBar
@OptIn(ExperimentalMaterial3AdaptiveApi::class)
@Composable
fun CompactOfficeShell(
    adaptiveInfo: WindowAdaptiveInfo,
    terminalBuffer: TerminalBuffer,
    onSendKey: (ByteArray) -> Unit
) {
    var selectedDestination by remember { mutableStateOf(OfficeDestination.COCKPIT) }
    val navSuiteType = NavigationSuiteScaffoldDefaults.navigationSuiteType(adaptiveInfo)

    NavigationSuiteScaffold(
        layoutType = navSuiteType,
        navigationSuiteItems = {
            OfficeDestination.values().forEach { dest ->
                item(
                    icon = { Icon(dest.icon, contentDescription = dest.label) },
                    label = { Text(dest.label) },
                    selected = dest == selectedDestination,
                    onClick = { selectedDestination = dest }
                )
            }
        }
    ) {
        when (selectedDestination) {
            OfficeDestination.COCKPIT -> TaskAndSubagentPane()
            OfficeDestination.CHAT -> ChatScreenPlaceholder()
            OfficeDestination.TERMINAL -> TerminalScreen(buffer = terminalBuffer, onSendKey = onSendKey)
            OfficeDestination.APPROVALS -> ApprovalsScreenPlaceholder()
        }
    }
}

// 3. Expanded: Desktop Mode / vivo Office Kit 3-Pane Cockpit
@OptIn(ExperimentalMaterial3AdaptiveApi::class, ExperimentalMaterial3AdaptiveLayoutApi::class)
@Composable
fun ExpandedOfficeShell(
    adaptiveInfo: WindowAdaptiveInfo,
    terminalBuffer: TerminalBuffer,
    onSendKey: (ByteArray) -> Unit
) {
    val navSuiteType = NavigationSuiteScaffoldDefaults.navigationSuiteType(adaptiveInfo)

    NavigationSuiteScaffold(
        layoutType = navSuiteType,
        navigationSuiteItems = {
            OfficeDestination.values().forEach { dest ->
                item(
                    icon = { Icon(dest.icon, contentDescription = dest.label) },
                    label = { Text(dest.label) },
                    selected = dest == OfficeDestination.COCKPIT,
                    onClick = {}
                )
            }
        }
    ) {
        SupportingPaneScaffold(
            mainPane = {
                // Left Column: Task list & subagents (280dp)
                Surface(
                    modifier = Modifier.widthIn(min = 260.dp, max = 320.dp).fillMaxHeight(),
                    color = MaterialTheme.colorScheme.surfaceContainer
                ) {
                    TaskAndSubagentPane()
                }
            },
            supportingPane = {
                // Center Column: Interactive terminal & Git diffs (Flexible weight)
                Box(
                    modifier = Modifier.fillMaxSize().background(Color.Black)
                ) {
                    TerminalScreen(buffer = terminalBuffer, onSendKey = onSendKey)
                }
            },
            extraPane = {
                // Right Column: Mac system vitals & token spend telemetry (320dp)
                Surface(
                    modifier = Modifier.widthIn(min = 300.dp, max = 360.dp).fillMaxHeight(),
                    color = MaterialTheme.colorScheme.surfaceContainer
                ) {
                    SystemVitalsAndPreviewPane()
                }
            }
        )
    }
}
```

### 4. Mouse & Trackpad Handling in Desktop Mode

Pointer events (mouse clicks, trackpad scrolls) route transparently through the same pointer input system. To support desktop-specific gestures like **Ctrl + Mouse Wheel to zoom terminal font size**:

```kotlin
package com.bimax.mobile.ui.desktop

import androidx.compose.foundation.layout.Box
import androidx.compose.runtime.Composable
import androidx.compose.ui.Modifier
import androidx.compose.ui.input.pointer.pointerInput

@Composable
fun TerminalZoomablePane(
    fontSize: Float,
    onFontSizeChange: (Float) -> Unit,
    content: @Composable () -> Unit
) {
    Box(
        modifier = Modifier.pointerInput(Unit) {
            awaitPointerEventScope {
                while (true) {
                    val event = awaitPointerEvent()
                    val scroll = event.changes.firstOrNull()?.scrollDelta?.y ?: 0f
                    if (scroll != 0f) {
                        // Scroll down zooms in, scroll up zooms out
                        val newSize = fontSize + (scroll * -0.5f)
                        onFontSizeChange(newSize.coerceIn(10f, 26f))
                    }
                }
            }
        }
    ) {
        content()
    }
}
```
