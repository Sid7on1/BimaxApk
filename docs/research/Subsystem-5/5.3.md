### 1. Verification Summary & Offline Resilience Architecture

Mobile cellular connections are inherently intermittent (subway cell handoffs, elevator fades, roaming switches). A remote developer control plane cannot afford to drop commands, lose approval decisions, or re-execute actions multiple times:

1. **Relational Offline Room Database**:
   - Persists tasks, chat transcripts, approval records, hardware vitals, and git diff hunks locally with foreign keys and cascading deletes.
   - All client-generated mutations are committed locally first (`isDirty = true`).
2. **Durable Outbox Pattern (`pending_commands`)**:
   - User commands are immediately written to an append-only `pending_commands` SQLite table with an auto-incrementing ID and a unique `idempotencyKey` (UUIDv4).
   - Prevents duplicate remote command execution across cellular reconnects.
3. **Reactive StateFlow Connection FSM**:
   - Models the socket lifecycle cleanly: `Disconnected` $\rightarrow$ `Connecting` $\rightarrow$ `Authenticating` $\rightarrow$ `Connected` $\rightarrow$ `Reconnecting`.
4. **Full-Jitter Exponential Backoff**:
   - Avoids thundering-herd reconnect storms on Cloudflare edge relays:
     $$T_{\text{sleep}} = \text{random}\Big(0, \min\big(M_{\text{max}}, B_{\text{base}} \cdot 2^{\text{attempt}}\big)\Big)$$
   - Resets backoff counter immediately upon reaching the `Connected` state.
5. **Sequential Outbox Flusher (`CommandFlusher`)**:
   - Drains the queue in strict monotonic `id ASC` order only while in the `Connected` state.
   - Deletes entries upon server ACK; halts flushing and marks `FAILED` on transmission drop.

```
+-----------------------------------------------------------------------------------------+
|                                OFFLINE-FIRST DATA PIPELINE                              |
|                                                                                         |
|  [User Actions: Approvals, Terminal Input, Chat Prompts]                                |
|           │                                                                             |
|           ▼                                                                             |
|  [1. Local Optimistic Write (Room Database)]                                            |
|     - Tasks, Messages, ApprovalRequests marked isDirty = true                           |
|     - Durable Outbox: INSERT INTO pending_commands (idempotencyKey, payload, status)    |
|           │                                                                             |
|           ▼                                                                             |
|  [2. Reactive Connection FSM: StateFlow<ConnectionState>]                               |
|     - Disconnected -> Connecting -> Authenticating -> Connected                         |
|     - On Error: Reconnecting with Full Jitter Exponential Backoff                       |
|           │                                                                             |
|           ▼ (Transition to Connected)                                                   |
|  [3. Sequential Command Flusher (CommandFlusher)]                                       |
|     - Queries pending_commands ORDER BY id ASC                                          |
|     - Dispatches encrypted envelope: {"sessionId", "key", "payload"}                    |
|     - Awaits Server ACK -> DELETE FROM pending_commands WHERE id = :id                  |
+-----------------------------------------------------------------------------------------+
```

```kotlin
package com.bimax.mobile.data.db

import androidx.room.*
import java.time.Instant

enum class TaskStatus { OPEN, IN_PROGRESS, DONE, CANCELLED }
enum class ApprovalStatus { PENDING, APPROVED, REJECTED }
enum class CommandStatus { PENDING, SENT, ACKED, FAILED }

// 1. Task Entity
@Entity(
    tableName = "tasks",
    indices = [Index("remoteId"), Index("updatedAt")]
)
data class TaskEntity(
    @PrimaryKey(autoGenerate = true) val id: Long = 0L,
    val remoteId: String? = null,
    val title: String,
    val description: String? = null,
    val status: TaskStatus = TaskStatus.OPEN,
    val isDirty: Boolean = false,
    val lastSyncedAt: Instant? = null,
    val createdAt: Instant = Instant.now(),
    val updatedAt: Instant = Instant.now()
)

// 2. Message Entity
@Entity(
    tableName = "messages",
    indices = [Index("taskId"), Index("createdAt")],
    foreignKeys = [
        ForeignKey(
            entity = TaskEntity::class,
            parentColumns = ["id"],
            childColumns = ["taskId"],
            onDelete = ForeignKey.CASCADE
        )
    ]
)
data class MessageEntity(
    @PrimaryKey(autoGenerate = true) val id: Long = 0L,
    val remoteId: String? = null,
    val taskId: Long,
    val senderId: String,
    val content: String,
    val createdAt: Instant = Instant.now(),
    val isPending: Boolean = false,
    val sequence: Long? = null
)

// 3. Approval Request Entity
@Entity(
    tableName = "approval_requests",
    indices = [Index("taskId"), Index("status")],
    foreignKeys = [
        ForeignKey(
            entity = TaskEntity::class,
            parentColumns = ["id"],
            childColumns = ["taskId"],
            onDelete = ForeignKey.CASCADE
        )
    ]
)
data class ApprovalRequestEntity(
    @PrimaryKey(autoGenerate = true) val id: Long = 0L,
    val remoteId: String? = null,
    val taskId: Long,
    val requestedByUserId: String,
    val status: ApprovalStatus = ApprovalStatus.PENDING,
    val createdAt: Instant = Instant.now(),
    val decidedAt: Instant? = null
)

// 4. System Vital Snapshot Entity
@Entity(
    tableName = "system_vital_snapshots",
    indices = [Index("timestamp")]
)
data class SystemVitalSnapshotEntity(
    @PrimaryKey(autoGenerate = true) val id: Long = 0L,
    val timestamp: Instant = Instant.now(),
    val batteryPercent: Int? = null,
    val isOnWifi: Boolean? = null,
    val isForeground: Boolean? = null,
    val cpuLoadPercent: Double? = null,
    val memoryUsedMb: Double? = null,
    val lastHeartbeatAgoMs: Long? = null,
    val connectionState: String? = null
)

// 5. Git Diff Hunk Entity
@Entity(
    tableName = "git_diff_hunks",
    indices = [Index("filePath"), Index("commitSha")]
)
data class GitDiffHunkEntity(
    @PrimaryKey(autoGenerate = true) val id: Long = 0L,
    val commitSha: String,
    val filePath: String,
    val oldStartLine: Int,
    val oldLineCount: Int,
    val newStartLine: Int,
    val newLineCount: Int,
    val patch: String,
    val taskId: Long? = null,
    val messageId: Long? = null
)

// 6. Durable Pending Command Outbox Entity
@Entity(
    tableName = "pending_commands",
    indices = [Index("createdAt"), Index("status")]
)
data class PendingCommandEntity(
    @PrimaryKey(autoGenerate = true) val id: Long = 0L,
    val createdAt: Instant = Instant.now(),
    val commandType: String,
    val payloadJson: String,
    val idempotencyKey: String,
    val status: CommandStatus = CommandStatus.PENDING,
    val lastAttemptAt: Instant? = null,
    val attemptCount: Int = 0
)

@Dao
interface PendingCommandDao {
    @Query("SELECT * FROM pending_commands WHERE status IN ('PENDING', 'FAILED') ORDER BY id ASC")
    suspend fun loadQueue(): List<PendingCommandEntity>

    @Insert
    suspend fun insert(command: PendingCommandEntity): Long

    @Update
    suspend fun update(command: PendingCommandEntity)

    @Query("DELETE FROM pending_commands WHERE id = :id")
    suspend fun deleteById(id: Long)
}

@Database(
    entities = [
        TaskEntity::class,
        MessageEntity::class,
        ApprovalRequestEntity::class,
        SystemVitalSnapshotEntity::class,
        GitDiffHunkEntity::class,
        PendingCommandEntity::class
    ],
    version = 1,
    exportSchema = false
)
@TypeConverters(Converters::class)
abstract class BimaxDatabase : RoomDatabase() {
    abstract fun pendingCommandDao(): PendingCommandDao
}

class Converters {
    @TypeConverter
    fun fromTimestamp(value: Long?): Instant? = value?.let { Instant.ofEpochMilli(it) }

    @TypeConverter
    fun dateToTimestamp(date: Instant?): Long? = date?.toEpochMilli()
}
```

### 3. StateFlow Connection FSM & Jittered Reconnector

```kotlin
package com.bimax.mobile.data.network

import kotlinx.coroutines.*
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlin.math.pow
import kotlin.random.Random

sealed class ConnectionState {
    data object Disconnected : ConnectionState()
    data class Connecting(val attempt: Int) : ConnectionState()
    data class Authenticating(val attempt: Int) : ConnectionState()
    data class Connected(val sessionId: String, val since: Long = System.currentTimeMillis()) : ConnectionState()
    data class Reconnecting(val attempt: Int, val nextRetryMs: Long) : ConnectionState()
}

interface WebSocketClient {
    suspend fun send(text: String)
    suspend fun awaitClosed()
    fun close(code: Int = 1000, reason: String = "Client shutting down")
}

/**
 * Full-Jitter Exponential Backoff:
 * delay = random(0, min(maxMs, baseMs * 2^attempt))
 */
fun reconnectDelayFullJitter(attempt: Int, baseMs: Long = 1000L, maxMs: Long = 30000L): Long {
    val exp = baseMs * 2.0.pow(attempt.coerceAtMost(16)).toLong()
    val capped = exp.coerceAtMost(maxMs)
    return (Random.nextDouble(0.0, 1.0) * capped).toLong()
}

class WebSocketManager(
    private val scope: CoroutineScope,
    private val wsFactory: suspend () -> WebSocketClient,
    private val authenticator: suspend (WebSocketClient) -> String,
    private val backoffStrategy: (Int) -> Long = ::reconnectDelayFullJitter
) {
    private val _state = MutableStateFlow<ConnectionState>(ConnectionState.Disconnected)
    val state: StateFlow<ConnectionState> = _state.asStateFlow()

    var currentClient: WebSocketClient? = null
        private set

    private var reconnectAttempt = 0
    private var reconnectJob: Job? = null

    fun start() {
        if (reconnectJob?.isActive == true) return
        reconnectJob = scope.launch(Dispatchers.IO) {
            reconnectAttempt = 0
            connectLoop()
        }
    }

    fun stop() {
        reconnectJob?.cancel()
        reconnectJob = null
        currentClient?.close()
        currentClient = null
        _state.value = ConnectionState.Disconnected
    }

    private suspend fun connectLoop() {
        while (coroutineContext.isActive) {
            _state.value = ConnectionState.Connecting(reconnectAttempt)

            try {
                val client = wsFactory()
                currentClient = client

                _state.value = ConnectionState.Authenticating(reconnectAttempt)
                val sessionId = authenticator(client)

                _state.value = ConnectionState.Connected(sessionId)
                reconnectAttempt = 0 // Reset backoff counter on successful connection

                client.awaitClosed()
            } catch (e: CancellationException) {
                _state.value = ConnectionState.Disconnected
                throw e
            } catch (e: Exception) {
                reconnectAttempt += 1
                val delayMs = backoffStrategy(reconnectAttempt)
                _state.value = ConnectionState.Reconnecting(reconnectAttempt, delayMs)
                delay(delayMs)
            }
        }
    }
}
```

### 4. Sequential Command Flusher (`CommandFlusher.kt`)

```kotlin
package com.bimax.mobile.data.network

import com.bimax.mobile.data.db.CommandStatus
import com.bimax.mobile.data.db.PendingCommandDao
import com.bimax.mobile.data.db.PendingCommandEntity
import kotlinx.coroutines.*
import java.time.Instant

class CommandFlusher(
    private val scope: CoroutineScope,
    private val pendingDao: PendingCommandDao,
    private val wsManager: WebSocketManager,
    private val ackTimeoutMs: Long = 5000L
) {
    private var flushJob: Job? = null

    fun startObserving() {
        flushJob?.cancel()
        flushJob = scope.launch(Dispatchers.IO) {
            wsManager.state.collect { state ->
                if (state is ConnectionState.Connected) {
                    flushPendingQueue(state.sessionId)
                }
            }
        }
    }

    private suspend fun flushPendingQueue(sessionId: String) {
        val commands = pendingDao.loadQueue()

        for (cmd in commands) {
            // Verify connection remains active before transmitting
            val currentState = wsManager.state.value
            if (currentState !is ConnectionState.Connected) break

            val now = Instant.now()
            val toSend = cmd.copy(
                status = CommandStatus.SENT,
                lastAttemptAt = now,
                attemptCount = cmd.attemptCount + 1
            )
            pendingDao.update(toSend)

            try {
                val client = wsManager.currentClient ?: break
                val envelopeJson = buildEnvelope(sessionId, toSend)
                client.send(envelopeJson)

                // Await server acknowledgment or timeout
                withTimeout(ackTimeoutMs) {
                    waitForServerAck(toSend.idempotencyKey)
                }

                // Delete successfully acknowledged command from database
                pendingDao.deleteById(cmd.id)
            } catch (e: Exception) {
                // Mark as failed; will retry automatically upon next reconnection
                pendingDao.update(toSend.copy(status = CommandStatus.FAILED))
                break
            }
        }
    }

    private suspend fun waitForServerAck(idempotencyKey: String) {
        // Coroutine channel or flow completion triggered by server ACK packet
        delay(10) // Mock representation of instantaneous ACK
    }

    private fun buildEnvelope(sessionId: String, cmd: PendingCommandEntity): String {
        return """
            {
                "session_id": "$sessionId",
                "command_type": "${cmd.commandType}",
                "idempotency_key": "${cmd.idempotencyKey}",
                "payload": ${cmd.payloadJson}
            }
        """.trimIndent()
    }
}
```
